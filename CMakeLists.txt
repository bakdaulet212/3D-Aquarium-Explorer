cmake_minimum_required(VERSION 3.21)
project(AquariumExplorer VERSION 2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
if(UNIX AND NOT APPLE)
  set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory(vendor/glfw EXCLUDE_FROM_ALL)
add_library(glad STATIC vendor/glad/src/glad.c)
target_include_directories(glad PUBLIC vendor/glad/include)
add_executable(AquariumExplorer src/main.cpp)
target_include_directories(AquariumExplorer PRIVATE vendor/glm)
target_link_libraries(AquariumExplorer PRIVATE glfw glad ${CMAKE_DL_LIBS})
target_compile_definitions(AquariumExplorer PRIVATE GLFW_INCLUDE_NONE)
if(MSVC)
  target_compile_options(AquariumExplorer PRIVATE /W4 /utf-8)
else()
  target_compile_options(AquariumExplorer PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_command(TARGET AquariumExplorer POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:AquariumExplorer>/shaders")
enable_testing()
add_test(NAME aquarium_math COMMAND AquariumExplorer --self-test)
